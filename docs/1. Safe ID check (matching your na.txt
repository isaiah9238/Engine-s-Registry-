// 1. Safe ID check (matching your name regex style)
function isValidId(id) {
  return id is string && id.size() > 0 && id.size() <= 128 && id.matches('^[a-zA-Z0-9_-]+$');
}

// 2. Safe Get Rule (using .get() so missing fields don't throw)
allow get: if isValidId(skillId) && (
  resource != null && (
    resource.data.get('isPublic', false) == true ||
    resource.data.get('status', '') == 'published' ||
    (isSignedIn() && (resource.data.authorId == request.auth.uid || isAdmin()))
  )
);

// 3. Safe Update Rule (safely compares createdAt even if null/omitted)
allow update: if isSignedIn()
  && isValidId(skillId)
  && (isAdmin() || (existing().authorId == request.auth.uid && incoming().authorId == existing().authorId))
  && incoming().get('createdAt', null) == existing().get('createdAt', null)
  && isValidSkill(incoming());
